## 1. Study Overview

This analysis estimates changes in frailty trajectories **before and after an incident cancer diagnosis** using a cohort-wide mixed-effects model.

- **Time since baseline** captures the background frailty trajectory (aging / secular changes).
- A **time-varying post-cancer indicator** estimates the **short-term level change** (the “jump”) after diagnosis.
- **Time since cancer** estimates the **change in slope** after diagnosis.

Key features:
- We include **both incident cancer cases and participants who never develop cancer** (no baseline cancer).
- Frailty outcome is an FI **excluding cancer items** (`fi_score_nocancer`).
- Visualization windows are data-driven but bounded to prevent modeling negative pre-baseline time.

## 2. Data Preparation & Eligibility


``` r
# Load data (imputed longitudinal FI dataset)
fi_long <- readRDS("/n/home06/xyzhou/Frailty/2_data analysis/before and after cancer/data/FI_longitudinal_1986_2020_IMPUTED_Cancer.rds")

# Study cycles used for repeated frailty measurement
target_cycles <- c("88", "92", "96", "00", "04", "08", "12", "16", "20")

# 1) Baseline covariates (Cycle 86)
baseline_covs <- fi_long %>%
  filter(cycle == "86") %>%
  select(
    id,
    age_months   = age_enrollment,
    base_race    = race,
    base_marital = marital_status,
    base_living  = living_arr,
    base_pckgr   = pckgr
  ) %>%
  mutate(base_age = age_months / 12) %>%
  select(-age_months) %>%
  drop_na()

# 2) Prepare longitudinal repeated-measure dataset (target cycles)
fi_prep <- fi_long %>%
  filter(cycle %in% target_cycles) %>%
  mutate(
    # Convert cycle code to calendar year (e.g., 88 -> 1988; 20 -> 2020)
    Year_num = case_when(
      as.numeric(cycle) > 50 ~ 1900 + as.numeric(cycle),
      TRUE ~ 2000 + as.numeric(cycle)
    ),
    Age = (worked_rtmnyr - dbmy09) / 12
  )

# 3) Define each participant’s baseline observation year in the analytic window
baseline_year_by_id <- fi_prep %>%
  group_by(id) %>%
  summarise(Year_base = min(Year_num), .groups = "drop")

# 4) Exclude baseline cancer (cancer present at first analytic observation)
baseline_cancer_ids <- fi_prep %>%
  inner_join(baseline_year_by_id, by = "id") %>%
  filter(Year_num == Year_base) %>%
  filter(fi_cancer == 1) %>%
  pull(id) %>%
  unique()

# 5) Identify incident cancer timing among those without baseline cancer
cancer_timing <- fi_prep %>%
  filter(!id %in% baseline_cancer_ids) %>%
  filter(fi_cancer == 1) %>%
  group_by(id) %>%
  summarise(Year_Cancer = min(Year_num), .groups = "drop")

# 6) Build analytic longitudinal dataset
analysis_long <- fi_prep %>%
  filter(!id %in% baseline_cancer_ids) %>%
  inner_join(baseline_year_by_id, by = "id") %>%
  left_join(cancer_timing, by = "id") %>%
  inner_join(baseline_covs, by = "id") %>%
  mutate(
    # Outcome validity rule 
    fi_score_nocancer = if_else(n_answered_nocancer < 20, NA_real_, fi_score_nocancer),

    # Follow-up time since baseline observation (years)
    time = Year_num - Year_base,

    # Time-varying post-cancer indicator
    postCancer = if_else(!is.na(Year_Cancer) & Year_num >= Year_Cancer, 1L, 0L),

    # Time since diagnosis (0 before diagnosis; increases after)
    timeAfterCancer = if_else(postCancer == 1L, Year_num - Year_Cancer, 0),

    # Useful indicators for tables/plots
    everCancer = if_else(!is.na(Year_Cancer), 1L, 0L),

    # Factor covariates
    base_race    = factor(base_race),
    base_marital = factor(base_marital),
    base_living  = factor(base_living),
    base_pckgr   = factor(base_pckgr),

    # Scale baseline age for stability
    base_age_z = as.numeric(scale(base_age))
  ) %>%
  filter(participated == 1, !is.na(fi_score_nocancer)) %>%
  group_by(id) %>%
  filter(n() >= 2) %>%
  ungroup() %>%
  arrange(id, Year_num)

4. Main Piecewise Linear Mixed-Effects Models

  fi_score_nocancer ~ time + postCancer + timeAfterCancer +
    base_age_z + base_race + base_marital + base_living + base_pckgr +
    (1 + time | id),
  data = analysis_long,
  control = lmerControl(optimizer = "bobyqa", optCtrl = list(maxfun = 1e5))
)

5. Visualization: Predicted Trajectories (Bounded Event-Centered)

We visualize the trajectories based on the data’s true follow-up span. To prevent the pre-cancer modeled time from becoming negative and generating a flatline, we bound the minimum relative plotting time against -tDx_median.

# Compute each cancer case's time-from-baseline to diagnosis
dx_time_by_id <- analysis_long %>%
  filter(everCancer == 1L) %>%
  distinct(id, Year_base, Year_Cancer) %>%
  mutate(tDx = Year_Cancer - Year_base)

tDx_median <- median(dx_time_by_id$tDx, na.rm = TRUE)

# Calculate the TRUE maximum and minimum relative follow-up times
analysis_long_cases <- analysis_long %>%
  filter(everCancer == 1L) %>%
  mutate(tRel = Year_num - Year_Cancer)

emp_min_rel <- floor(min(analysis_long_cases$tRel, na.rm = TRUE))
emp_max_rel <- ceiling(max(analysis_long_cases$tRel, na.rm = TRUE))

# BOUND min_rel to prevent negative 'time' values in the model
min_rel <- max(emp_min_rel, -floor(tDx_median))
max_rel <- emp_max_rel

cat("Plotting follow-up range from", min_rel, "to", max_rel, "years relative to diagnosis.\n")

y_post_jump_ext <- grid_cancer_post_plot_ext$pred_FI[grid_cancer_post_plot_ext$tRel == 0]

# Generate Plot
ggplot() +
  # Plot Never Cancer trajectory (Unique intercept and slope)
  geom_line(data = grid_never_plot_ext, aes(x = tRel, y = pred_FI), 
            color = "#E69F00", linetype = "dashed", linewidth = 1.2) +
  
  # Plot Pre-Cancer trajectory
  geom_line(data = grid_cancer_pre_plot_ext, aes(x = tRel, y = pred_FI), 
            color = "#0072B2", linewidth = 1.2) +
  
  # Plot Post-Cancer trajectory
  geom_line(data = grid_cancer_post_plot_ext, aes(x = tRel, y = pred_FI), 
            color = "#0072B2", linewidth = 1.2) +
  
  # Add the vertical "jump" at Time 0
  geom_segment(aes(x = 0, xend = 0, 
                   y = y_pre_jump_ext, 
                   yend = y_post_jump_ext), 
               color = "#0072B2", linewidth = 1.2) +
  
  geom_vline(xintercept = 0, linetype = "dotted", color = "gray30") +
  
  scale_x_continuous(breaks = scales::pretty_breaks(n = 10)) +
  theme_minimal() +
  theme(
    panel.grid.minor = element_blank(),
    panel.border = element_rect(color = "black", fill = NA, linewidth = 0.5),
    plot.title = element_text(face = "bold", size = 15),
    legend.position = "bottom"
  ) +
  labs(
    title = "Predicted FI Trajectories: Testing Pre-Diagnosis Differences",
    subtitle = paste0("Extended model allowing unique baseline frailty and pre-diagnosis slopes (tDx = ", round(tDx_median, 1), ")."),
    x = "Years relative to cancer diagnosis (0 = Diagnosis)",
    y = "Predicted non-cancer FI"
  )
